//go:build e2e

package postgres_test

import (
	"context"
	"os"
	"testing"

	v1 "NYCU-SDC/caravanserai/api/v1"
	"NYCU-SDC/caravanserai/internal/event"
	"NYCU-SDC/caravanserai/internal/store"
	pgstore "NYCU-SDC/caravanserai/internal/store/postgres"

	"github.com/stretchr/testify/assert"
	"github.com/stretchr/testify/require"
	"go.uber.org/zap"
)

func TestFencedStatusReportRejectsRecreatedProjectAtSameVersion(t *testing.T) {
	ctx := context.Background()
	var s *pgstore.Store
	var bus *event.Bus
	// An explicitly supplied disposable database also permits running this
	// database test in environments without Docker. Otherwise use the shared
	// helper to start and clean up a PostgreSQL container.
	if databaseURL := os.Getenv("CARA_FENCED_CAS_TEST_DATABASE_URL"); databaseURL != "" {
		bus = event.New(zap.NewNop(), 64)
		var err error
		s, err = pgstore.New(ctx, databaseURL, zap.NewNop(), bus)
		require.NoError(t, err)
		t.Cleanup(s.Close)
	} else {
		s, bus = newE2EStore(t)
	}

	const name = "fenced-status-recreated"
	old := createRunning(t, s, name, "node-a", 1)
	t.Cleanup(func() { require.NoError(t, s.DeleteProject(ctx, name)) })
	ref := store.AssignmentRef{
		UID:        old.ObjectMeta.UID,
		NodeRef:    old.Status.NodeRef,
		Generation: old.Status.AssignmentGeneration,
	}
	events := bus.Subscribe(event.TopicProjectUpdated)
	var replacement *v1.Project
	mutations := 0

	err := s.ReportProjectStatusFenced(ctx, name, ref, func(status *v1.ProjectStatus) error {
		mutations++
		require.Equal(t, 1, mutations, "a stale assignment must not be retried")
		// This callback runs after the store reads the fenced snapshot, but
		// before it swaps the status. Recreate synchronously to fix the
		// ordering without goroutines, sleeps, or scheduler assumptions.
		require.NoError(t, s.DeleteProject(ctx, name))
		replacement = createRunning(t, s, name, "node-a", 1)
		require.NotEqual(t, old.ObjectMeta.UID, replacement.ObjectMeta.UID)
		require.Equal(t, old.ObjectMeta.ResourceVersion, replacement.ObjectMeta.ResourceVersion,
			"the replacement must have the same version to expose a version-only swap")
		drain(events)
		status.Phase = v1.ProjectPhaseFailed
		return nil
	})

	assert.ErrorIs(t, err, store.ErrStaleAssignment,
		"a report for the deleted lifetime must not be accepted by its same-name successor")
	require.Equal(t, 1, mutations)
	require.NotNil(t, replacement)
	after, err := s.GetProject(ctx, name)
	require.NoError(t, err)
	assert.Equal(t, replacement.ObjectMeta, after.ObjectMeta, "a rejected report must not update metadata")
	assert.Equal(t, replacement.Status, after.Status, "a rejected report must not overwrite the replacement's status")
	assert.Zero(t, drain(events), "a rejected report must not publish a status update")
}
